    // set up vertex data: one vertex per pixel of the height map, its height from the pixel's value
    // ---------------------------------------------------------------------------------------------
    std::vector<float> vertices;
    float yScale = 64.0f / 256.0f, yShift = 16.0f; // apply a scale and a shift to the height data
    unsigned bytePerPixel = nrChannels;
    for (int i = 0; i < height; i++)
    {
        for (int j = 0; j < width; j++)
        {
            unsigned char* pixelOffset = data + (j + width * i) * bytePerPixel;
            unsigned char y = pixelOffset[0];

            // vertex: column j along x, row i along z (the image's top row at the back)
            vertices.push_back(-width / 2.0f + width * j / (float)width);   // vx
            vertices.push_back((int)y * yScale - yShift);                    // vy
            vertices.push_back(-height / 2.0f + height * i / (float)height); // vz
        }
    }
    m_vertexCount = static_cast<UINT>(vertices.size() / 3);
    stbi_image_free(data);

    // Create the vertex buffer: the terrain's vertices, 3 floats each.
    {
        const UINT vertexBufferSize = static_cast<UINT>(vertices.size() * sizeof(float));

        // An upload heap, as in the earlier chapters (simple, not the fastest for the GPU).
        CD3DX12_HEAP_PROPERTIES heapProperties(D3D12_HEAP_TYPE_UPLOAD);
        CD3DX12_RESOURCE_DESC bufferDesc = CD3DX12_RESOURCE_DESC::Buffer(vertexBufferSize);
        ThrowIfFailed(m_device->CreateCommittedResource(
            &heapProperties,
            D3D12_HEAP_FLAG_NONE,
            &bufferDesc,
            D3D12_RESOURCE_STATE_GENERIC_READ,
            nullptr,
            IID_PPV_ARGS(&m_vertexBuffer)));

        // Copy the vertex data to the vertex buffer.
        UINT8* pVertexDataBegin = nullptr;
        CD3DX12_RANGE readRange(0, 0);        // We do not intend to read from this resource on the CPU.
        ThrowIfFailed(m_vertexBuffer->Map(0, &readRange, reinterpret_cast<void**>(&pVertexDataBegin)));
        memcpy(pVertexDataBegin, vertices.data(), vertexBufferSize);
        m_vertexBuffer->Unmap(0, nullptr);

        // Initialize the vertex buffer view.
        m_vertexBufferView.BufferLocation = m_vertexBuffer->GetGPUVirtualAddress();
        m_vertexBufferView.StrideInBytes = 3 * sizeof(float);
        m_vertexBufferView.SizeInBytes = vertexBufferSize;
    }
