=== task: ex-fading
--- file: main.cpp
--- find:
    constantBufferData.light.diffuse = XMFLOAT3(0.8f, 0.8f, 0.8f);
    constantBufferData.light.specular = XMFLOAT3(1.0f, 1.0f, 1.0f);
--- replace:
    // a failing battery: the flashlight fades out and in again
    static auto startTime = std::chrono::high_resolution_clock::now();
    auto currentTime = std::chrono::high_resolution_clock::now();
    float time = std::chrono::duration<float, std::chrono::seconds::period>(currentTime - startTime).count();
    float strength = 0.5f + 0.5f * std::sin(time * 2.0f);
    constantBufferData.light.diffuse = XMFLOAT3(0.8f * strength, 0.8f * strength, 0.8f * strength);
    constantBufferData.light.specular = XMFLOAT3(strength, strength, strength);
