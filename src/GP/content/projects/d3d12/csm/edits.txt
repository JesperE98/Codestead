// The light-space matrix of the part of the camera's frustum between nearPlane and farPlane:
// a light view looking at the part's centre along the light's direction, and an orthographic
// projection around its eight corners.
XMFLOAT4X4 D3D12App::getLightSpaceMatrix(const float nearPlane, const float farPlane)
{
    const XMMATRIX proj = XMMatrixPerspectiveFovRH(XMConvertToRadians(fov), static_cast<float>(m_width) / static_cast<float>(m_height), nearPlane, farPlane);
    const XMMATRIX view = XMMatrixLookAtRH(cameraPos, cameraPos + cameraFront, cameraUp);
    const std::vector<XMFLOAT4> corners = getFrustumCornersWorldSpace(proj, view);

    XMVECTOR center = XMVectorZero();
    for (const XMFLOAT4& v : corners)
    {
        center += XMLoadFloat4(&v);
    }
    center /= static_cast<float>(corners.size());

    const XMMATRIX lightView = XMMatrixLookAtRH(center + lightDir, center, XMVectorSet(0.0f, 1.0f, 0.0f, 0.0f));

    // the corners' bounding box in light space (-FLT_MAX: the lowest float)
    float minX = FLT_MAX;
    float maxX = -FLT_MAX;
    float minY = FLT_MAX;
    float maxY = -FLT_MAX;
    float minZ = FLT_MAX;
    float maxZ = -FLT_MAX;
    for (const XMFLOAT4& v : corners)
    {
        const XMVECTOR trf = XMVector4Transform(XMLoadFloat4(&v), lightView);
        minX = fminf(minX, XMVectorGetX(trf));
        maxX = fmaxf(maxX, XMVectorGetX(trf));
        minY = fminf(minY, XMVectorGetY(trf));
        maxY = fmaxf(maxY, XMVectorGetY(trf));
        minZ = fminf(minZ, XMVectorGetZ(trf));
        maxZ = fmaxf(maxZ, XMVectorGetZ(trf));
    }

    // Tune this parameter according to the scene: the depth range reaches further towards and
    // away from the light, so objects outside the camera's view still cast their shadows into it
    constexpr float zMult = 10.0f;
    if (minZ < 0)
    {
        minZ *= zMult;
    }
    else
    {
        minZ /= zMult;
    }
    if (maxZ < 0)
    {
        maxZ /= zMult;
    }
    else
    {
        maxZ *= zMult;
    }

    // learnopengl's glm::ortho(minX, maxX, minY, maxY, minZ, maxZ): the same box, with
    // Direct3D's depth from 0 to 1
    const XMMATRIX lightProjection = XMMatrixOrthographicOffCenterRH(minX, maxX, minY, maxY, minZ, maxZ);
    XMFLOAT4X4 lightSpaceMatrix;
    XMStoreFloat4x4(&lightSpaceMatrix, lightView * lightProjection);    // row vectors: the view first, then the projection
    return lightSpaceMatrix;
}

// one light-space matrix per cascade: camera near plane .. level 0, level 0 .. level 1, ...,
// last level .. camera far plane
std::vector<XMFLOAT4X4> D3D12App::getLightSpaceMatrices()
{
    std::vector<XMFLOAT4X4> ret;
    for (size_t i = 0; i < shadowCascadeLevels.size() + 1; ++i)
    {
        if (i == 0)
        {
            ret.push_back(getLightSpaceMatrix(cameraNearPlane, shadowCascadeLevels[i]));
        }
        else if (i < shadowCascadeLevels.size())
        {
            ret.push_back(getLightSpaceMatrix(shadowCascadeLevels[i - 1], shadowCascadeLevels[i]));
        }
        else
        {
            ret.push_back(getLightSpaceMatrix(shadowCascadeLevels[i - 1], cameraFarPlane));
        }
    }
    return ret;
}

// The eight corners of a frustum in world space: the corners of the clip-space box (x and y
// from -1 to 1, Direct3D's z from 0 to 1) taken back through the inverse of view * projection.
// (In containers DirectXMath keeps its storage types: XMFLOAT4, XMFLOAT4X4.)
std::vector<XMFLOAT4> getFrustumCornersWorldSpace(const XMMATRIX& projview)
{
    const XMMATRIX inv = XMMatrixInverse(nullptr, projview);

    std::vector<XMFLOAT4> frustumCorners;
    for (unsigned int x = 0; x < 2; ++x)
    {
        for (unsigned int y = 0; y < 2; ++y)
        {
            for (unsigned int z = 0; z < 2; ++z)
            {
                const XMVECTOR pt = XMVector4Transform(XMVectorSet(2.0f * x - 1.0f, 2.0f * y - 1.0f, static_cast<float>(z), 1.0f), inv);
                XMFLOAT4 corner;
                XMStoreFloat4(&corner, pt / XMVectorSplatW(pt));
                frustumCorners.push_back(corner);
            }
        }
    }

    return frustumCorners;
}

std::vector<XMFLOAT4> getFrustumCornersWorldSpace(const XMMATRIX& proj, const XMMATRIX& view)
{
    return getFrustumCornersWorldSpace(view * proj);    // row vectors: the view first
}

float ShadowCalculation(float3 fragPosWorldSpace, float3 normal)
{
    // select cascade layer
    int layer = CascadeLayer(fragPosWorldSpace);

    float4 fragPosLightSpace = mul(float4(fragPosWorldSpace, 1.0f), lightSpaceMatrices[layer]);
    // perform perspective divide
    float3 projCoords = fragPosLightSpace.xyz / fragPosLightSpace.w;
    // transform x and y from [-1,1] to [0,1] texture coordinates: in Direct3D v points down
    // while y points up, so y is flipped. z needs nothing: Direct3D's depth is 0..1 already.
    float2 shadowUV = float2(projCoords.x * 0.5f + 0.5f, -projCoords.y * 0.5f + 0.5f);

    // get depth of current fragment from light's perspective
    float currentDepth = projCoords.z;

    // keep the shadow at 0.0 when outside the far_plane region of the light's frustum.
    if (currentDepth > 1.0f)
    {
        return 0.0f;
    }
    // calculate bias (based on depth map resolution and slope), smaller for the bigger cascades
    float bias = max(0.05f * (1.0f - dot(normal, lightDir)), 0.005f);
    const float biasModifier = 0.5f;
    if (layer == cascadeCount)
    {
        bias *= 1.0f / (farPlane * biasModifier);
    }
    else
    {
        bias *= 1.0f / (cascadePlaneDistances[layer].x * biasModifier);
    }

    // get closest depth value from light's perspective (using [0,1] range fragPosLight as coords)
    float closestDepth = shadowMap.Sample(shadowSampler, float3(shadowUV, layer)).r;
    // check whether current frag pos is in shadow
    float shadow = (currentDepth - bias) > closestDepth ? 1.0f : 0.0f;

    return shadow;
}
