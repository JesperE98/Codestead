// The pixel shader: the colour of the window's pixel.
float4 PSMain(PSInput input) : SV_TARGET
{
    // The 3x3 samples around the pixel. v grows downwards in Direct3D, so the top row
    // is at -offset (learnopengl's GLSL, where v grows upwards, has +offset there).
    float2 offsets[9] =
    {
        float2(-offset, -offset),    // top-left
        float2( 0.0f,   -offset),    // top-center
        float2( offset, -offset),    // top-right
        float2(-offset,  0.0f),      // center-left
        float2( 0.0f,    0.0f),      // center-center
        float2( offset,  0.0f),      // center-right
        float2(-offset,  offset),    // bottom-left
        float2( 0.0f,    offset),    // bottom-center
        float2( offset,  offset)     // bottom-right
    };

    // The kernel: how much each sample counts.
    float kernel[9] =
    {
        1.0f,  1.0f, 1.0f,
        1.0f, -8.0f, 1.0f,
        1.0f,  1.0f, 1.0f
    };

    float3 sampleTex[9];
    for (int i = 0; i < 9; i++)
    {
        sampleTex[i] = screenTexture.Sample(wrapSampler, input.texCoord + offsets[i]).rgb;
    }
    float3 col = float3(0.0f, 0.0f, 0.0f);
    for (int k = 0; k < 9; k++)
        col += sampleTex[k] * kernel[k];

    return float4(col, 1.0f);
}

// Utility function for loading a cube map from six image files, in the order +X (right),
// -X (left), +Y (top), -Y (bottom), +Z (front), -Z (back): creates one texture with six
// array slices (the faces) in a default heap and its shader resource view (TEXTURECUBE) in
// slot descriptorIndex of the CBV/SRV heap, and records the copy of the pixels into it.
// The copy reads from uploadHeap, so that must stay alive until the command list has executed.
void D3D12App::LoadCubemap(const std::vector<std::string>& faces, ComPtr<ID3D12Resource>& texture, ComPtr<ID3D12Resource>& uploadHeap, UINT descriptorIndex)
{
    // Load the six images as RGBA. No vertical flip: a face's first row is its top
    // (learnopengl also loads its cube map without the flip).
    int width = 0, height = 0, nrChannels = 0;
    unsigned char* data[6] = {};
    for (UINT i = 0; i < 6; i++)
    {
        data[i] = stbi_load(faces[i].c_str(), &width, &height, &nrChannels, 4);
        if (data[i] == nullptr)
        {
            throw std::runtime_error("Cubemap texture failed to load at path: " + faces[i]);
        }
    }

    // Describe and create the texture: like a 2D texture, but with 6 array slices
    // (DepthOrArraySize), one per face, all the same size. It starts in the COPY_DEST state.
    D3D12_RESOURCE_DESC textureDesc = {};
    textureDesc.MipLevels = 1;
    textureDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM;
    textureDesc.Width = static_cast<UINT64>(width);
    textureDesc.Height = static_cast<UINT>(height);
    textureDesc.Flags = D3D12_RESOURCE_FLAG_NONE;
    textureDesc.DepthOrArraySize = 6;
    textureDesc.SampleDesc.Count = 1;
    textureDesc.SampleDesc.Quality = 0;
    textureDesc.Dimension = D3D12_RESOURCE_DIMENSION_TEXTURE2D;

    CD3DX12_HEAP_PROPERTIES heapProperties(D3D12_HEAP_TYPE_DEFAULT);
    ThrowIfFailed(m_device->CreateCommittedResource(
        &heapProperties,
        D3D12_HEAP_FLAG_NONE,
        &textureDesc,
        D3D12_RESOURCE_STATE_COPY_DEST,
        nullptr,
        IID_PPV_ARGS(&texture)));

    // The upload heap: big enough for all six subresources (the faces) as the GPU lays them out.
    const UINT64 uploadBufferSize = GetRequiredIntermediateSize(texture.Get(), 0, 6);

    CD3DX12_HEAP_PROPERTIES uploadHeapProperties(D3D12_HEAP_TYPE_UPLOAD);
    CD3DX12_RESOURCE_DESC uploadBufferDesc = CD3DX12_RESOURCE_DESC::Buffer(uploadBufferSize);
    ThrowIfFailed(m_device->CreateCommittedResource(
        &uploadHeapProperties,
        D3D12_HEAP_FLAG_NONE,
        &uploadBufferDesc,
        D3D12_RESOURCE_STATE_GENERIC_READ,
        nullptr,
        IID_PPV_ARGS(&uploadHeap)));

    // One D3D12_SUBRESOURCE_DATA per face: subresource i is array slice i, the face
    // GL_TEXTURE_CUBE_MAP_POSITIVE_X + i in OpenGL. UpdateSubresources copies all six into
    // the upload heap now and schedules the copy into the texture.
    D3D12_SUBRESOURCE_DATA faceData[6] = {};
    for (UINT i = 0; i < 6; i++)
    {
        faceData[i].pData = data[i];
        faceData[i].RowPitch = width * 4;
        faceData[i].SlicePitch = faceData[i].RowPitch * height;
    }
    UpdateSubresources(m_commandList.Get(), texture.Get(), uploadHeap.Get(), 0, 0, 6, faceData);

    // After the copy, the texture is read by the pixel shader.
    CD3DX12_RESOURCE_BARRIER barrier = CD3DX12_RESOURCE_BARRIER::Transition(texture.Get(), D3D12_RESOURCE_STATE_COPY_DEST, D3D12_RESOURCE_STATE_PIXEL_SHADER_RESOURCE);
    m_commandList->ResourceBarrier(1, &barrier);

    // Describe and create a SRV that sees the six slices as one cube (a TextureCube in HLSL).
    D3D12_SHADER_RESOURCE_VIEW_DESC srvDesc = {};
    srvDesc.Shader4ComponentMapping = D3D12_DEFAULT_SHADER_4_COMPONENT_MAPPING;
    srvDesc.Format = textureDesc.Format;
    srvDesc.ViewDimension = D3D12_SRV_DIMENSION_TEXTURECUBE;
    srvDesc.TextureCube.MostDetailedMip = 0;
    srvDesc.TextureCube.MipLevels = 1;
    CD3DX12_CPU_DESCRIPTOR_HANDLE srvHandle(m_cbvSrvHeap->GetCPUDescriptorHandleForHeapStart(), descriptorIndex, m_cbvSrvDescriptorSize);
    m_device->CreateShaderResourceView(texture.Get(), &srvDesc, srvHandle);

    // Free the image memory (UpdateSubresources has copied it into the upload heap).
    for (UINT i = 0; i < 6; i++)
    {
        stbi_image_free(data[i]);
    }
}
