// Up and Down change the mix value, in steps of 0.1 (holding a key repeats it).
void D3D12App::OnKeyDown(UINT8 key)
{
    if (key == VK_UP)
    {
        m_mixValue += 0.1f;
        if (m_mixValue >= 1.0f)
            m_mixValue = 1.0f;
    }
    if (key == VK_DOWN)
    {
        m_mixValue -= 0.1f;
        if (m_mixValue <= 0.0f)
            m_mixValue = 0.0f;
    }
}
