=== task: ex-fading
--- file: main.cpp
--- find:
        lightingShader.setVec3("light.diffuse", 0.8f, 0.8f, 0.8f);
        lightingShader.setVec3("light.specular", 1.0f, 1.0f, 1.0f);
--- replace:
        // a failing battery: the flashlight fades out and in again
        float strength = static_cast<float>(0.5 + 0.5 * sin(glfwGetTime() * 2.0));
        lightingShader.setVec3("light.diffuse", glm::vec3(0.8f) * strength);
        lightingShader.setVec3("light.specular", glm::vec3(1.0f) * strength);
