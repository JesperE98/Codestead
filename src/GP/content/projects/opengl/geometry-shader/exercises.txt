{

=== task: ex-face-normals
--- file: normal_visualization.gs
--- find:
void main()
{
    GenerateLine(0); // first vertex normal
    GenerateLine(1); // second vertex normal
    GenerateLine(2); // third vertex normal
}
--- replace:
void main()
{
    // one line per triangle: from its centre along the face's normal
    vec4 centre = (gl_in[0].gl_Position + gl_in[1].gl_Position + gl_in[2].gl_Position) / 3.0;
    vec3 a = vec3(gl_in[0].gl_Position) - vec3(gl_in[1].gl_Position);
    vec3 b = vec3(gl_in[2].gl_Position) - vec3(gl_in[1].gl_Position);
    vec3 normal = normalize(cross(b, a)); // outwards for a counter-clockwise triangle
    gl_Position = projection * centre;
    EmitVertex();
    gl_Position = projection * (centre + vec4(normal, 0.0) * MAGNITUDE);
    EmitVertex();
    EndPrimitive();
}
