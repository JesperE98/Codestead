=== task: ex-invert-specular
--- file: lighting_maps.fs
--- find:
    vec3 specular = light.specular * spec * texture(material.specular, TexCoords).rgb;
--- replace:
    vec3 specular = light.specular * spec * (1.0 - texture(material.specular, TexCoords).rgb); // the wood shines, the steel not

=== task: ex-emission-mask
--- file: lighting_maps.fs
--- find:
    vec3 emission = texture(material.emission, TexCoords).rgb;
--- replace:
    // only on the wood, where the specular map is (nearly) black
    float onWood = texture(material.specular, TexCoords).r < 0.1 ? 1.0 : 0.0;
    vec3 emission = texture(material.emission, TexCoords).rgb * onWood;

=== task: ex-scroll-emission
--- file: lighting_maps.fs
--- find:
uniform Light light;
--- replace:
uniform Light light;
uniform float time;
--- file: lighting_maps.fs
--- find:
    vec3 emission = texture(material.emission, TexCoords).rgb;
--- replace:
    vec3 emission = texture(material.emission, TexCoords + vec2(0.0, time)).rgb; // the letters move one face height per second
--- file: main.cpp
--- find:
        lightingShader.setFloat("material.shininess", 64.0f);

        lightingShader.setFloat("time", static_cast<float>(glfwGetTime()));
